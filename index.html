📁 Kết Nối Google Drive (Thư Mục Tổng: HỒ SƠ LƯU VIỆT PHÚ)

Nếu team đã có sẵn folder HỒ SƠ LƯU VIỆT PHÚ dùng chung, bạn có thể dán ID trên thanh địa chỉ Drive vào đây (hoặc để trống để hệ thống tự tạo).

Mã Google Apps Script chuẩn 3 cấp thư mục: 
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var parentFolder;
  if (data.rootFolderId) {
    parentFolder = DriveApp.getFolderById(data.rootFolderId);
  } else {
    var rootName = "HỒ SƠ LƯU VIỆT PHÚ";
    var rootSearch = DriveApp.getRootFolder().getFoldersByName(rootName);
    parentFolder = rootSearch.hasNext() ? rootSearch.next() : DriveApp.getRootFolder().createFolder(rootName);
  }
  var folders = parentFolder.getFoldersByName(data.folderName);
  var projectFolder = folders.hasNext() ? folders.next() : parentFolder.createFolder(data.folderName);
  var subNames = ["1. File word", "2. File scan", "3. Kết quả", "4. Hợp đồng pháp lý"];
  var subMap = {};
  subNames.forEach(function(sn) {
    var existing = projectFolder.getFoldersByName(sn);
    subMap[sn] = existing.hasNext() ? existing.next() : projectFolder.createFolder(sn);
  });
  var out = [];
  data.files.forEach(function(f) {
    var targetFolder = subMap[f.category] || projectFolder;
    var blob = Utilities.newBlob(Utilities.base64Decode(f.base64), f.mimeType, f.name);
    var file = targetFolder.createFile(blob);
    out.push({ name: f.name, category: f.category, url: file.getUrl(), downloadUrl: file.getDownloadUrl() });
  });
  return ContentService.createTextOutput(JSON.stringify({ ok: true, folderUrl: projectFolder.getUrl(), files: out })).setMimeType(ContentService.MimeType.JSON);
}
