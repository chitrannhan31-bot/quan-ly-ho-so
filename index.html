📁 Kết Nối Google Drive (1 Dự Án = 1 Folder Con Tự Động)

Upload file Word, PDF, Kết quả, Hợp đồng sẽ tự động tạo folder con trên Drive

Hướng dẫn cài 1 lần duy nhất (2 phút):

1. Tạo 1 thư mục mẹ trên Google Drive (VD: HO_SO_LUU_TEAM_CHI), mở thư mục đó và copy ID thư mục trên thanh địa chỉ (đoạn mã sau /folders/).

2. Vào script.google.com ➔ Dự án mới ➔ dán đoạn mã bên dưới (thay ID thư mục mẹ của anh/chị) ➔ bấm Triển khai (Deploy) ➔ Tùy chọn triển khai mới ➔ Ứng dụng web (Web app), chọn quyền truy cập Bất kỳ ai (Anyone) rồi copy link Web App dán vào ô dưới.

Mã Google Apps Script cầu nối Drive: 
function doPost(e) {
  var PARENT_FOLDER_ID = "DAN_ID_FOLDER_ME_CUA_BAN_VAO_DAY";
  var data = JSON.parse(e.postData.contents);
  var parent = DriveApp.getFolderById(PARENT_FOLDER_ID);
  var folders = parent.getFoldersByName(data.folderName);
  var subFolder = folders.hasNext() ? folders.next() : parent.createFolder(data.folderName);
  var savedFiles = [];
  (data.files || []).forEach(function(f) {
    var blob = Utilities.newBlob(Utilities.base64Decode(f.base64), f.mimeType, f.name);
    var file = subFolder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    savedFiles.push({
      name: f.name,
      category: f.category,
      size: f.size,
      url: file.getUrl(),
      downloadUrl: "https://drive.google.com/uc?export=download&id=" + file.getId()
    });
  });
  return ContentService.createTextOutput(JSON.stringify({
    ok: true,
    folderUrl: subFolder.getUrl(),
    files: savedFiles
  })).setMimeType(ContentService.MimeType.JSON);
}
